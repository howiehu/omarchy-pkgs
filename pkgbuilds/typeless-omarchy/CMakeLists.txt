cmake_minimum_required(VERSION 3.21)
project(typeless-omarchy VERSION 0.4.1 LANGUAGES CXX)
find_package(Fcitx5Core REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
add_library(typeless-voice MODULE src/voice.cpp)
target_compile_features(typeless-voice PRIVATE cxx_std_20)
target_compile_options(typeless-voice PRIVATE -Wall -Wextra -Werror)
target_link_libraries(typeless-voice PRIVATE Fcitx5::Core PkgConfig::JSONC PkgConfig::GLIB)
set_target_properties(typeless-voice PROPERTIES PREFIX "")
install(TARGETS typeless-voice DESTINATION lib/fcitx5)
install(FILES data/typeless-voice-addon.conf DESTINATION share/fcitx5/addon RENAME typeless-voice.conf)
install(PROGRAMS bin/typeless-omarchy bin/typeless-omarchy-setup DESTINATION bin)
install(PROGRAMS runtime/prepare.py DESTINATION lib/typeless-omarchy/runtime)
install(PROGRAMS helpers/gdbus DESTINATION lib/typeless-omarchy/helpers)
install(FILES data/typeless.lua DESTINATION share/typeless-omarchy)
install(FILES LICENSE DESTINATION share/licenses/typeless-omarchy)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME runtime-archive COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime.py)
  add_test(NAME protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.py)
endif()
